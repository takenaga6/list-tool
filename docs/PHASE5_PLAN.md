### Step 5: 相互作用ボーナス関数の実装（30分）

追加する関数:

def check_interaction_bonus(signals: dict) -> tuple[int, str]:
    """相互作用ボーナス判定
    
    49社データで継続率100%だった3点セット、または5個以上保有 → +3点ボーナス
    
    signals: 各シグナルの判定結果 dict
        {
            's1_pr': bool,
            's2_kenko_media': bool,
            's3_welfare': bool,
            's4_kenko_keiei': bool,
            's5_renewal': bool,
            's6_jisha_bldg': bool,
            's7_iso': bool,
            's8_sdgs': bool,
            's9_president_health': bool,
            's10_profitable': bool,
        }
    
    Returns:
        (bonus_points, reason_text)
        bonus_points: 0 or 3（重複加算しない）
    """
    # 3点セット定義（49社データで継続率100%だった組み合わせ）
    # 注: S10 は新規シグナルで49社データには直接ない
    #     代わりに既存シグナルの組み合わせで判定
    triple_sets = [
        ('s4_kenko_keiei', 's7_iso', 's10_profitable'),  # 健康経営+ISO+儲かってる業界
        ('s4_kenko_keiei', 's7_iso', 's8_sdgs'),         # 健康経営+ISO+SDGs (4/4=100%)
        ('s4_kenko_keiei', 's7_iso', 's9_president_health'),  # 健康経営+ISO+社長詳細 (4/4=100%)
        ('s7_iso', 's9_president_health', 's10_profitable'),  # ISO+社長詳細+儲かってる業界
        ('s7_iso', 's9_president_health', 's3_welfare'),  # ISO+社長詳細+福利厚生 (4/4=100%)
        ('s4_kenko_keiei', 's8_sdgs', 's10_profitable'),  # 健康経営+SDGs+儲かってる業界
    ]
    
    # 3点セット完成チェック
    for triple in triple_sets:
        if all(signals.get(s, False) for s in triple):
            return (3, f"3点セット達成: {'+'.join(triple)}")
    
    # シグナル5個以上保有チェック
    total = sum(1 for v in signals.values() if v)
    if total >= 5:
        return (3, f"シグナル{total}個保有")
    
    return (0, "")
